5. Source Code

departures.c — Main C Program

/* Enable POSIX extensions (required for select, etc.) */
#define _POSIX_C_SOURCE 200112L

/*
 * departures.c - TTC Transit Terminal Departure Board
 *
 * Simulates a real-time TTC (Toronto Transit Commission) departure board.
 * Reads route and schedule data from routes.dat and displays a live-
 * refreshing countdown board styled in TTC colours (red and white).
 *
 * Features:
 *   - Live terminal UI that refreshes every 500 ms (no Enter key needed)
 *   - Countdown timers showing the next 2 departures per route
 *   - HH:MM scheduled time displayed alongside each countdown
 *   - Seconds-level countdown for arrivals under 2 minutes
 *   - Service Alerts: random Delay / Stalled statuses per session
 *   - Direction toggle: press 'd' to cycle All / Eastbound+Northbound /
 *     Westbound+Southbound views
 *   - Colour-coded urgency: green > 5 min, yellow 2-5 min, red < 2 min
 *
 * Build:  gcc -o departures departures.c -lncurses
 * Run:    ./departures
 *
 * Data file: routes.dat (must be in the same directory)
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <ncurses.h>
#include <unistd.h>
#include <sys/select.h>
#include <ctype.h>

/* ── Constants ────────────────────────────────────────────────────────── */
#define MAX_ROUTES        64
#define MAX_DEPARTURES   200
#define MAX_NAME_LEN      64
#define DATA_FILE    "routes.dat"
#define REFRESH_MS       500
#define SHOW_NEXT          2
#define ALERT_CHANCE      15
#define DELAY_MIN_MINS     3
#define DELAY_MAX_MINS    12

/* Column layout */
#define COL_ROUTE_NUM     1
#define COL_ROUTE_NAME    6
#define COL_DIRECTION    24
#define COL_DEP1         38
#define COL_DEP2         51
#define COL_STATUS       65
#define COL_WIDTH_DEP    12
#define MIN_COLS         80
#define MIN_ROWS         10

/* Colour pair IDs */
#define CLR_HEADER      1
#define CLR_SUBHEADER   2
#define CLR_ROUTE_NUM   3
#define CLR_ROUTE_NAME  4
#define CLR_TIME_GREEN  5
#define CLR_TIME_YELLOW 6
#define CLR_TIME_RED    7
#define CLR_ALERT_DELAY 8
#define CLR_ALERT_STALL 9
#define CLR_BORDER     10
#define CLR_FOOTER     11
#define CLR_DIM        12
#define CLR_TIME_NOW   13

typedef enum { STATUS_ON_TIME = 0, STATUS_DELAYED, STATUS_STALLED } ServiceStatus;

typedef struct {
    char route_num[16];
    char route_name[MAX_NAME_LEN];
    char direction[32];
    char stop_name[MAX_NAME_LEN];
    int  departures[MAX_DEPARTURES];
    int  num_departures;
    ServiceStatus status;
    int  delay_minutes;
} Route;

static Route routes[MAX_ROUTES];
static int   num_routes = 0;
static int   direction_filter = 0;
static const char *filter_labels[] = {
    "ALL ROUTES", "EASTBOUND / NORTHBOUND", "WESTBOUND / SOUTHBOUND"
};

/* Parse HH:MM string into minutes from midnight */
static int hhmm_to_minutes(const char *hhmm) {
    int h = 0, m = 0;
    if (sscanf(hhmm, "%d:%d", &h, &m) != 2) return -1;
    return h * 60 + m;
}

/* Format minutes-from-midnight as compact 12-hour string, e.g. "4:47p" */
static void minutes_to_hhmm12(int min, char *buf, int len) {
    int t = min % 1440; if (t < 0) t += 1440;
    int h = t / 60, m = t % 60;
    char ap = (h < 12) ? 'a' : 'p';
    if (h == 0) h = 12; else if (h > 12) h -= 12;
    snprintf(buf, len, "%d:%02d%c", h, m, ap);
}

/* Load routes from data file, generating timetables from headway data */
static int parse_routes(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (!fp) return 0;
    char line[512];
    while (fgets(line, sizeof(line), fp) && num_routes < MAX_ROUTES) {
        char *p = line;
        while (*p == ' ' || *p == '\t') p++;
        if (*p == '#' || *p == '\n' || *p == '\r' || *p == '\0') continue;
        char rn[16], rname[64], dir[32], sname[64], s[16], e[16];
        int hw = 0, off = 0;
        if (sscanf(p, "%15[^|]|%63[^|]|%31[^|]|%63[^|]|%15[^|]|%15[^|]|%d|%d",
                   rn, rname, dir, sname, s, e, &hw, &off) != 8 || hw <= 0) continue;
        Route *r = &routes[num_routes];
        snprintf(r->route_num,  sizeof(r->route_num),  "%s", rn);
        snprintf(r->route_name, sizeof(r->route_name), "%s", rname);
        snprintf(r->direction,  sizeof(r->direction),  "%s", dir);
        snprintf(r->stop_name,  sizeof(r->stop_name),  "%s", sname);
        int start = hhmm_to_minutes(s), end = hhmm_to_minutes(e);
        if (start < 0 || end < 0) continue;
        if (end < start) end += 1440;
        r->num_departures = 0;
        for (int t = start + off; t <= end && r->num_departures < MAX_DEPARTURES; t += hw)
            r->departures[r->num_departures++] = t % 1440;
        r->status = STATUS_ON_TIME; r->delay_minutes = 0;
        num_routes++;
    }
    fclose(fp);
    return num_routes;
}

/* Randomly assign service alerts (ALERT_CHANCE% of routes) */
static void assign_alerts(void) {
    srand((unsigned int)time(NULL));
    for (int i = 0; i < num_routes; i++) {
        if (rand() % 100 < ALERT_CHANCE) {
            if (rand() % 3 == 0) { routes[i].status = STATUS_STALLED; }
            else {
                routes[i].status = STATUS_DELAYED;
                routes[i].delay_minutes = DELAY_MIN_MINS +
                    rand() % (DELAY_MAX_MINS - DELAY_MIN_MINS + 1);
            }
        }
    }
}

static int route_matches_filter(const Route *r) {
    if (direction_filter == 0) return 1;
    if (direction_filter == 1)
        return strncmp(r->direction,"East",4)==0 || strncmp(r->direction,"North",5)==0;
    return strncmp(r->direction,"West",4)==0 || strncmp(r->direction,"South",5)==0;
}

/* Get the next 'count' departure times after now_seconds */
static int get_next_departures(const Route *r, long now_secs,
                               int *out, int count) {
    int found = 0; long nm = now_secs / 60;
    for (int w = 0; w <= 1 && found < count; w++) {
        long base = (long)w * 1440;
        for (int i = 0; i < r->num_departures && found < count; i++) {
            long t = (long)r->departures[i] + base + r->delay_minutes;
            if (t > nm) out[found++] = (int)(t % 1440);
        }
    }
    return found;
}

/* Build countdown string "4m (4:47p)" or "45s (4:43p)" or "** NOW **" */
static int format_departure_cell(int dep_min, long now_secs,
                                  char *buf, int len) {
    long ds = (long)dep_min * 60, diff = ds - now_secs;
    if (diff < -43200) diff += 86400; if (diff > 43200) diff -= 86400;
    char ts[10]; minutes_to_hhmm12(dep_min, ts, sizeof(ts));
    if (diff <= 0)  { snprintf(buf,len,"** NOW **   "); return CLR_TIME_NOW; }
    if (diff < 60)  { snprintf(buf,len,"%lds (%s) ",diff,ts); return CLR_TIME_RED; }
    if (diff < 120) { snprintf(buf,len,"1m (%s) ",ts);        return CLR_TIME_RED; }
    if (diff <= 300){ snprintf(buf,len,"%ldm (%s) ",diff/60,ts); return CLR_TIME_YELLOW; }
    snprintf(buf,len,"%ldm (%s) ",diff/60,ts); return CLR_TIME_GREEN;
}

static void init_ncurses(void) {
    initscr(); cbreak(); noecho();
    keypad(stdscr, TRUE); nodelay(stdscr, TRUE); curs_set(0);
    start_color(); use_default_colors();
    init_pair(CLR_HEADER,     COLOR_WHITE,  COLOR_RED);
    init_pair(CLR_SUBHEADER,  COLOR_WHITE,  COLOR_RED);
    init_pair(CLR_ROUTE_NUM,  COLOR_YELLOW, COLOR_BLACK);
    init_pair(CLR_ROUTE_NAME, COLOR_WHITE,  COLOR_BLACK);
    init_pair(CLR_TIME_GREEN, COLOR_GREEN,  COLOR_BLACK);
    init_pair(CLR_TIME_YELLOW,COLOR_YELLOW, COLOR_BLACK);
    init_pair(CLR_TIME_RED,   COLOR_RED,    COLOR_BLACK);
    init_pair(CLR_ALERT_DELAY,COLOR_YELLOW, COLOR_RED);
    init_pair(CLR_ALERT_STALL,COLOR_WHITE,  COLOR_RED);
    init_pair(CLR_BORDER,     COLOR_RED,    COLOR_BLACK);
    init_pair(CLR_FOOTER,     COLOR_BLACK,  COLOR_WHITE);
    init_pair(CLR_DIM,        COLOR_WHITE,  COLOR_BLACK);
    init_pair(CLR_TIME_NOW,   COLOR_RED,    COLOR_BLACK);
}

static void render(int tick) {
    int rows, cols; getmaxyx(stdscr, rows, cols);
    if (cols < MIN_COLS || rows < MIN_ROWS) {
        clear(); mvprintw(0,0,"Terminal too small."); refresh(); return;
    }
    clear();
    time_t now_t = time(NULL); struct tm *lt = localtime(&now_t);
    long now_secs = (long)lt->tm_hour*3600 + lt->tm_min*60 + lt->tm_sec;
    const char *sname = num_routes ? routes[0].stop_name : "Unknown";

    /* Header */
    attron(COLOR_PAIR(CLR_HEADER)|A_BOLD);
    for(int c=0;c<cols;c++){mvaddch(0,c,' ');mvaddch(1,c,' ');}
    mvprintw(0,1," TTC  TORONTO TRANSIT COMMISSION");
    char clk[20];
    strftime(clk,sizeof(clk),(tick%2==0)?"%I:%M:%S %p":"%I %M %S %p",lt);
    mvprintw(0, cols-(int)strlen(clk)-2, "%s", clk);
    mvprintw(1,1," STOP: %-28s", sname);
    const char *filt=filter_labels[direction_filter];
    mvprintw(1,cols-(int)strlen(filt)-2,"%s",filt);
    attroff(COLOR_PAIR(CLR_HEADER)|A_BOLD);

    /* Column headers */
    attron(COLOR_PAIR(CLR_SUBHEADER)|A_BOLD);
    for(int c=0;c<cols;c++) mvaddch(2,c,' ');
    mvprintw(2,COL_ROUTE_NUM,"%-4s","RTE");
    mvprintw(2,COL_ROUTE_NAME,"%-17s","ROUTE NAME");
    mvprintw(2,COL_DIRECTION,"%-13s","DIRECTION");
    mvprintw(2,COL_DEP1,"%-12s","NEXT");
    mvprintw(2,COL_DEP2,"%-12s","FOLLOWING");
    mvprintw(2,COL_STATUS,"%-15s","STATUS");
    attroff(COLOR_PAIR(CLR_SUBHEADER)|A_BOLD);

    int dr=3, vis=0;
    for(int i=0;i<num_routes;i++) {
        if(!route_matches_filter(&routes[i])) continue;
        if(dr+2>rows-2) break;
        /* Separator */
        attron(COLOR_PAIR(CLR_BORDER));
        for(int c=0;c<cols;c++) mvaddch(dr,c,'-');
        attroff(COLOR_PAIR(CLR_BORDER));
        dr++;
        /* Route data */
        attron(COLOR_PAIR(CLR_ROUTE_NUM)|A_BOLD);
        mvprintw(dr,COL_ROUTE_NUM,"%-4s",routes[i].route_num);
        attroff(COLOR_PAIR(CLR_ROUTE_NUM)|A_BOLD);
        attron(COLOR_PAIR(CLR_ROUTE_NAME));
        mvprintw(dr,COL_ROUTE_NAME,"%-17s",routes[i].route_name);
        mvprintw(dr,COL_DIRECTION,"%-13s",routes[i].direction);
        attroff(COLOR_PAIR(CLR_ROUTE_NAME));
        int nxt[SHOW_NEXT], found=get_next_departures(&routes[i],now_secs,nxt,SHOW_NEXT);
        int dc[2]={COL_DEP1,COL_DEP2};
        for(int j=0;j<SHOW_NEXT;j++) {
            if(j<found) {
                char cell[32]; int clr=format_departure_cell(nxt[j],now_secs,cell,32);
                attron(COLOR_PAIR(clr)|((clr==CLR_TIME_RED||clr==CLR_TIME_NOW)?A_BOLD:0));
                mvprintw(dr,dc[j],"%-12s",cell);
                attroff(COLOR_PAIR(clr)|A_BOLD);
            } else {
                attron(COLOR_PAIR(CLR_DIM)|A_DIM);
                mvprintw(dr,dc[j],"%-12s","---");
                attroff(COLOR_PAIR(CLR_DIM)|A_DIM);
            }
        }
        if(routes[i].status==STATUS_STALLED) {
            attron(COLOR_PAIR(CLR_ALERT_STALL)|A_BOLD|A_BLINK);
            mvprintw(dr,COL_STATUS,"%-15s","** STALLED **");
            attroff(COLOR_PAIR(CLR_ALERT_STALL)|A_BOLD|A_BLINK);
        } else if(routes[i].status==STATUS_DELAYED) {
            char ds[20]; snprintf(ds,sizeof(ds),"DELAY +%d min",routes[i].delay_minutes);
            attron(COLOR_PAIR(CLR_ALERT_DELAY)|A_BOLD);
            mvprintw(dr,COL_STATUS,"%-15s",ds);
            attroff(COLOR_PAIR(CLR_ALERT_DELAY)|A_BOLD);
        } else {
            attron(COLOR_PAIR(CLR_TIME_GREEN));
            mvprintw(dr,COL_STATUS,"%-15s","On Time");
            attroff(COLOR_PAIR(CLR_TIME_GREEN));
        }
        dr+=1; vis++;
    }
    attron(COLOR_PAIR(CLR_BORDER));
    for(int c=0;c<cols;c++) mvaddch(dr,c,'=');
    attroff(COLOR_PAIR(CLR_BORDER));
    attron(COLOR_PAIR(CLR_FOOTER)|A_BOLD);
    for(int c=0;c<cols;c++) mvaddch(rows-1,c,' ');
    mvprintw(rows-1,1," [D] Toggle Direction   [Q] Quit   Showing %d route(s)",vis);
    attroff(COLOR_PAIR(CLR_FOOTER)|A_BOLD);
    refresh();
}

int main(void) {
    if (parse_routes(DATA_FILE) == 0) {
        fprintf(stderr,"Error: cannot read '%s'\n", DATA_FILE);
        return EXIT_FAILURE;
    }
    assign_alerts();
    init_ncurses();
    int running=1, tick=0;
    while(running) {
        render(tick++);
        int ch=getch();
        if(tolower(ch)=='q') running=0;
        else if(tolower(ch)=='d') direction_filter=(direction_filter+1)%3;
        struct timeval tv={0, REFRESH_MS*1000};
        select(0,NULL,NULL,NULL,&tv);
    }
    endwin();
    printf("Thanks for riding with TTC!\n");
    return EXIT_SUCCESS;
}

Errors and How They Were Fixed

The AI made several errors that it caught and corrected autonomously:

  • nanosleep() not declared — The first version used nanosleep() which conflicted with -std=c11. The AI self-corrected by switching to POSIX select() with a timeval struct — a cleaner, more portable solution.
  • usleep / useconds_t not declared — An intermediate attempt with usleep() also failed; the AI tried adding _POSIX_C_SOURCE before settling on the select() approach.
  • Column overlap bug — The initial version placed 3 departure cells at columns 38, 49, and 60, but the status badge started at column 61, causing them to overlap. The AI identified this during the refinement pass and redesigned the layout to 2 departure cells with the status starting at column 65.
  • strncpy truncation compiler warnings — Addressed by replacing all strncpy() calls with snprintf().

No manual code edits were required. All corrections were autonomous once the error was described (either by the compiler or by the AI reviewing its own output).

What Was Easy and What Was Difficult

Easy: The initial prompt was straightforward — specifying the option (A), the language (C), and the platform (Linux/Matrix-compatible). Difficult: The GitHub token setup was the most painful part of the lab — not due to AI limitations but due to GitHub's fine-grained PAT permission model. Reflection on Agentic AI and My Career

The most striking observation from this lab is that the AI's relationship with me is collaborative, not competitive. The AI excels at producing syntactically correct, well-structured boilerplate at speed — but it required my judgment to specify the requirements, evaluate the output, catch the column overlap visually, and navigate the GitHub token permissions issue. The AI cannot test in a real terminal, see what the board looks like at runtime, or know that the token error message was misleading.

This suggests that as a software developer entering the industry, my value will shift from "writing code" toward "specifying, evaluating, and integrating AI-generated code." Skills like reading compiler errors, understanding system APIs (POSIX, ncurses), and diagnosing runtime behaviour remain essential — because the AI needs those skills from me when it gets stuck. The developer who understands what the AI produces (and when it's wrong) will consistently outperform one who treats AI output as a black box. Agentic AI is a powerful multiplier, but it multiplies the effectiveness of a developer who already has a foundation — not a substitute for one.
